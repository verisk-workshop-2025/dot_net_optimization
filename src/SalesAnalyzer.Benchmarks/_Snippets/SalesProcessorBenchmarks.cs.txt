using BenchmarkDotNet.Attributes;
using BenchmarkDotNet.Jobs;
using SalesAnalyzer.Lib.Models;
using SalesAnalyzer.Lib.Services;

namespace SalesAnalyzer.Benchmarks
{
    [SimpleJob(runtimeMoniker: RuntimeMoniker.Net80)]
    [MemoryDiagnoser]
    public class SalesProcessorBenchmarks
    {
        SalesProcessor processor;

        [GlobalSetup]
        public void Setup()
        {
            processor = new SalesProcessor();
        }

        [IterationCleanup]
        public void IterationCleanup()
        {
            processor.Branches.Clear();
        }


        [Benchmark(Baseline = true)]
        public Dictionary<string, Branch> InitializeUnoptimized()
        {
            processor.InitializeUnoptimized();

            return processor.Branches;
        }

        [Benchmark]
        public Dictionary<string, Branch> InitializeOptimized()
        {
            processor.InitializeOptimized();

            return processor.Branches;
        }
    }
}
